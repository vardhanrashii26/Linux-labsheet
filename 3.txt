cat > backup_check.sh <<'EOF'
#!/bin/bash
if [ $# -eq 0 ]; then
    echo "Usage: ./backup_check.sh <directory>"
    exit 1
fi
if [ ! -d "$1" ]; then
    echo "Directory does not exist"
    exit 1
fi
mkdir -p logs
count=0
while IFS= read -r -d '' file
do
    echo "Backup found: $file"
    count=$((count+1))
done < <(find "$1" -type f -name "*.bak" -print0)
echo "Total backup files: $count"
if [ "$count" -eq 0 ]; then
    echo "No backup files found" >> logs/backup_log.txt
fi
EOF
chmod +x backup_check.sh
touch backup/data1.bak backup/data2.bak
./backup_check.sh backup


cat > file_report.sh <<'EOF'
#!/bin/bash
if [ $# -ne 1 ]; then
    echo "Usage: ./file_report.sh <filename>"
    exit 1
fi
mkdir -p logs
if [ ! -f "$1" ]; then
    echo "Error: File does not exist"
    echo "$(date) - Failed: $1 does not exist" >> logs/execution_history.log
    exit 1
fi
if [ ! -r "$1" ]; then
    echo "Error: File is not readable"
    echo "$(date) - Failed: $1 is not readable" >> logs/execution_history.log
    exit 1
fi
echo "File Report: $1"
wc "$1"
echo "$(date) - Successfully processed: $1" >> logs/execution_history.log
echo "Processing completed successfully"
EOF
chmod +x file_report.sh

echo "Welcome to Bash" >> sample.txt
