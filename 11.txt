tar -xvf files.tar -C extracted_tar
ls -R extracted_tar

ls archive_lab

ls archive_lab

ls -lh archive.tar.gz
tar -tzvf archive.tar.gz

tar -xzvf archive.tar.gz -C extracted_targz
ls -R extracted_targz

ls -lh files.zip
unzip -l files.zip

unzip files.zip -d extracted_zip
ls -R extracted_zip
